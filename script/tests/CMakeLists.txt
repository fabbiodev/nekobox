cmake_minimum_required(VERSION 3.20)
project(NekoBoxTransitionTests LANGUAGES CXX)
enable_testing()
add_executable(profile-operation-queue profile-operation-queue.cpp)
target_compile_features(profile-operation-queue PRIVATE cxx_std_20)
target_include_directories(profile-operation-queue PRIVATE ../../src)
add_test(NAME profile-operation-queue COMMAND profile-operation-queue)
set_tests_properties(profile-operation-queue PROPERTIES TIMEOUT 10)
find_package(Qt6 QUIET COMPONENTS Core)
if(Qt6_FOUND)
    add_executable(responsive-task responsive-task.cpp)
    target_compile_features(responsive-task PRIVATE cxx_std_20)
    target_include_directories(responsive-task PRIVATE ../../src)
    target_link_libraries(responsive-task PRIVATE Qt6::Core)
    if(WIN32)
        add_custom_command(TARGET responsive-task POST_BUILD
            COMMAND ${CMAKE_COMMAND} -E copy_if_different
                $<TARGET_FILE:Qt6::Core> $<TARGET_FILE_DIR:responsive-task>)
    endif()
    add_test(NAME responsive-task COMMAND responsive-task)
    set_tests_properties(responsive-task PROPERTIES TIMEOUT 10)
endif()
